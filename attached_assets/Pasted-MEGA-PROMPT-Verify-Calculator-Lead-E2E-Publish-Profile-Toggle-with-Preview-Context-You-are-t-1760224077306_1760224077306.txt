Add (or confirm) in apps/app/server/routes.ts:

// Get current user's tenant profile (slug + publish state) – auth required
app.get('/api/me/profile-public', ensureAuthUnified, async (req, res) => {
  const tenantId = req.tenant.id;
  const row = await profilesRepo.getByTenantId(tenantId); // implement using your db layer
  // Shape
  return res.json({
    slug: row?.slug ?? null,
    isPublished: Boolean(row?.isPublished),
    name: row?.name ?? null,
  });
});

// Update slug & publish state – auth required
app.post('/api/me/profile/publish', ensureAuthUnified, async (req, res) => {
  const tenantId = req.tenant.id;
  const { slug, isPublished } = req.body ?? {};
  // 1) normalize slug (kebab-case, a–z0–9–, trimmed, 3–64 chars)
  function toSlug(s) {
    return String(s || '')
      .toLowerCase()
      .replace(/[^a-z0-9]+/g, '-')
      .replace(/^-+|-+$/g, '')
      .slice(0, 64);
  }
  const newSlug = slug ? toSlug(slug) : null;
  if (newSlug && newSlug.length < 3) return res.status(400).json({ error: 'slug_too_short' });

  // 2) uniqueness check (exclude this tenantId)
  if (newSlug) {
    const exists = await profilesRepo.existsBySlugExceptTenant(newSlug, tenantId);
    if (exists) return res.status(409).json({ error: 'slug_taken' });
  }

  // 3) persist fields (merge, not clobber)
  const updated = await profilesRepo.updatePublishSettings(tenantId, {
    slug: newSlug ?? undefined, // if null, don't overwrite existing
    isPublished: typeof isPublished === 'boolean' ? isPublished : undefined,
  });

  return res.json({
    ok: true,
    slug: updated.slug,
    isPublished: Boolean(updated.isPublished),
  });
});
