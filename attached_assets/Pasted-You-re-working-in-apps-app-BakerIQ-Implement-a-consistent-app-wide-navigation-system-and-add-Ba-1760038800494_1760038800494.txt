Shared App Layout with Consistent Menu
Make an AppLayout wrapper that renders the top nav and a content container. Use it on every page.

New: client/src/components/AppLayout.tsx

import { Link, useLocation } from 'wouter';
import { NAV_ITEMS } from '@/navigation/navConfig';

export default function AppLayout({ children }: { children: React.ReactNode }) {
  const [location] = useLocation();
  const demo = import.meta.env.VITE_DEMO_MODE === 'true';

  const items = NAV_ITEMS.filter(i => demo || !i.demoOnly);

  return (
    <div className="min-h-screen">
      {/* Top Nav */}
      <div className="border-b bg-white">
        <div className="max-w-6xl mx-auto px-4 h-14 flex items-center gap-4">
          <div className="font-semibold">BakerIQ</div>
          <nav className="flex gap-3">
            {items.map(it => {
              const active = location === it.href || location.startsWith(it.href + '/');
              return (
                <Link key={it.href} href={it.href} className={`px-2 py-1 rounded ${active ? 'bg-black text-white' : 'hover:bg-neutral-100'}`}>
                  {it.label}
                </Link>
              );
            })}
          </nav>
        </div>
      </div>
      {/* Page container */}
      <main className="max-w-6xl mx-auto px-4 py-6">{children}</main>
    </div>
  );
}


Back Button / Breadcrumb Helper
Add a shared header with a Back button you can drop into any page.

New: client/src/components/PageHeader.tsx

export function PageHeader({ title, subtitle }: { title: string; subtitle?: string }) {
  return (
    <div className="mb-4 flex items-center justify-between">
      <div>
        <h1 className="text-xl font-semibold">{title}</h1>
        {subtitle && <p className="text-sm opacity-70">{subtitle}</p>}
      </div>
      <button onClick={() => history.back()} className="px-3 py-1 rounded border hover:bg-neutral-50">
        ← Back
      </button>
    </div>
  );
}
