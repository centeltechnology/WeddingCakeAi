import crypto from 'crypto';
import { db } from './db';
import { customers, quotes, quoteItems, quoteEvents } from './db/schema'; // adjust
import { eq, and, desc } from 'drizzle-orm';
import { ensureAuthUnified } from './auth'; // adjust

// List recent customers (picker)
app.get('/api/customers', ensureAuthUnified, async (req, res) => {
  const tenantId = req.tenant.id;
  const rows = await db.select().from(customers)
    .where(eq(customers.tenantId, tenantId))
    .orderBy(desc(customers.updatedAt ?? customers.createdAt))
    .limit(100);
  res.json(rows);
});

// Upsert customer by email (fallback phone), returns row
app.post('/api/customers/upsert', ensureAuthUnified, async (req, res) => {
  const tenantId = req.tenant.id;
  const { name, email, phone } = req.body ?? {};
  if (!name && !email && !phone) return res.status(400).json({ error: 'invalid_customer' });

  let row: any = null;
  if (email) [row] = await db.select().from(customers).where(and(eq(customers.tenantId, tenantId), eq(customers.email, email))).limit(1);
  if (!row && phone) [row] = await db.select().from(customers).where(and(eq(customers.tenantId, tenantId), eq(customers.phone, phone))).limit(1);

  if (row) {
    await db.update(customers).set({
      name: name ?? row.name, phone: phone ?? row.phone,
    }).where(eq(customers.id, row.id));
  } else {
    const id = crypto.randomUUID();
    await db.insert(customers).values({
      id, tenantId, name: name ?? email ?? phone ?? 'Customer',
      email: email ?? null, phone: phone ?? null,
    });
  }

  const [out] = await db.select().from(customers)
    .where(email ? and(eq(customers.tenantId, tenantId), eq(customers.email, email))
                 : and(eq(customers.tenantId, tenantId), eq(customers.phone, phone)))
    .limit(1);
  res.json({ ok: true, customer: out });
});

// Save an estimate as a draft quote (+ items)
app.post('/api/estimates/save-as-quote', ensureAuthUnified, async (req, res) => {
  const tenantId = req.tenant.id;
  const { customer, title, items, taxRate, discount, depositPct, notes } = req.body ?? {};
  if (!Array.isArray(items) || items.length === 0) return res.status(400).json({ error: 'no_items' });

  // Upsert/resolve customerId
  let customerId = customer?.id as string | undefined;
  if (!customerId) {
    const r = await fetch(`http://localhost:${process.env.PORT ?? 5000}/api/customers/upsert`, {
      method: 'POST', headers: { 'Content-Type': 'application/json', cookie: req.headers.cookie ?? '' },
      body: JSON.stringify({ name: customer?.name, email: customer?.email, phone: customer?.phone })
    });
    const j = await r.json();
    if (!r.ok || !j?.customer?.id) return res.status(400).json({ error: 'customer_upsert_failed' });
    customerId = j.customer.id;
  }

  // Totals
  const subtotal = items.reduce((s: number, it: any) => s + (Number(it.qty) || 0) * (Number(it.price) || 0), 0);
  const discountAmt = Math.max(0, Number(discount) || 0);
  const taxRateN = Math.max(0, Number(taxRate) || 0);
  const taxed = taxRateN * Math.max(0, subtotal - discountAmt);
  const total = Math.max(0, subtotal - discountAmt + taxed);
  const depositPctN = Math.max(0, Math.min(1, Number(depositPct) || 0));

  const quoteId = crypto.randomUUID();
  await db.insert(quotes).values({
    id: quoteId, tenantId, customerId,
    title: title || 'Estimate', status: 'draft',
    subtotal, discount: discountAmt, tax: taxed, total, depositPct: depositPctN,
    notes: notes ?? null, createdAt: new Date()
  });

  for (const it of items) {
    await db.insert(quoteItems).values({
      id: crypto.randomUUID(), tenantId, quoteId,
      name: String(it.name ?? 'Item'), qty: Number(it.qty) || 1,
      unit: (it.unit ?? 'ea').toString(), price: Number(it.price) || 0,
      notes: it.notes ?? null,
    });
  }

  try {
    await db.insert(quoteEvents).values({
      id: crypto.randomUUID(), tenantId, quoteId, type: 'created',
      meta: { source: 'baker_calculator' }, createdAt: new Date()
    });
  } catch {}

  res.status(201).json({ ok: true, quoteId, totals: { subtotal, discount: discountAmt, tax: taxed, total, depositPct: depositPctN } });
});
