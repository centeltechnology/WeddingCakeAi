app.post('/api/me/profile', authGuard, async (req, res) => {
  const tenantId = req.tenant.id;
  const body = req.body ?? {};
  // allow flat fields and JSON buckets for social/payments
  const patch: any = {
    displayName: body.displayName ?? body.businessName ?? null,
    phone: body.phone ?? null,
    website: body.website ?? null,
    address: body.address ?? null,
    about: body.about ?? null,
    specialties: body.specialties ?? null,
    logoUrl: body.logoUrl ?? null,
    coverUrl: body.coverUrl ?? null,
  };

  // merge JSON objects instead of clobbering
  if (body.social && typeof body.social === 'object') {
    patch.social = sql`COALESCE(${tenantProfiles.social}, '{}'::jsonb) || ${JSON.stringify(body.social)}::jsonb`;
  }
  if (body.payments && typeof body.payments === 'object') {
    patch.payments = sql`COALESCE(${tenantProfiles.payments}, '{}'::jsonb) || ${JSON.stringify(body.payments)}::jsonb`;
  }

  const existing = await db.query.tenantProfiles.findFirst({ where: eq(tenantProfiles.tenantId, tenantId) });
  if (existing) {
    const [u] = await db.update(tenantProfiles)
      .set({ ...patch, updatedAt: new Date() })
      .where(eq(tenantProfiles.tenantId, tenantId))
      .returning();
    return res.json(u);
  } else {
    const [i] = await db.insert(tenantProfiles).values({
      id: crypto.randomUUID(),
      tenantId,
      ...patch
    }).returning();
    return res.json(i);
  }
});

import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';

export function useProfile() {
  return useQuery({
    queryKey: ['profile'],
    queryFn: async () => (await fetch('/api/me/profile')).json()
  });
}
export function useSaveProfile() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: async (data: any) => {
      const r = await fetch('/api/me/profile', {
        method: 'POST',
        headers: { 'Content-Type':'application/json' },
        body: JSON.stringify(data)
      });
      if (!r.ok) throw new Error('save_failed');
      return r.json();
    },
    onSuccess: () => qc.invalidateQueries({ queryKey: ['profile'] })
  });
}
