B) Single Slug→Tenant Resolver (source of truth)

Create apps/app/server/lib/tenantResolver.ts:

import { db } from '../db';
import { tenants, tenantProfiles } from '../db/schema'; // adjust imports
import { eq } from 'drizzle-orm';

export async function resolveTenantBySlug(slug: string){
  if (!slug) return null;
  // Prefer the profile table if that’s where slug lives
  const [row] = await db.select({
    id: tenants.id,
    slug: tenantProfiles.slug,
    isPublished: tenantProfiles.isPublished,
  })
  .from(tenants)
  .leftJoin(tenantProfiles, eq(tenantProfiles.tenantId, tenants.id))
  .where(eq(tenantProfiles.slug, slug))
  .limit(1);
  return row ?? null;
}


Use it in public routes (no auth required):

// Listing /p/:slug

  const found = await resolveTenantBySlug(req.params.slug);

D) Demo Tenant Publish (seed + verify)

Create apps/app/server/scripts/seedDemoTenant.ts (idempotent):

import { db } from '../db';
import { users, tenants, tenantProfiles } from '../db/schema';
import { eq } from 'drizzle-orm';
import crypto from 'crypto';

async function run(){
  const email = 'demo@bakeriq.app';
  const password = 'DemoPass123!';
  const slug = 'sweet-treats-bakery';

  // upsert user
  let [user] = await db.select().from(users).where(eq(users.email, email)).limit(1);
  if (!user){
    const id = crypto.randomUUID();
    await db.insert(users).values({ id, email, passwordHash: await hash(password), verified: true });
    [user] = await db.select().from(users).where(eq(users.email, email)).limit(1);
  } else if (!user.verified) {
    await db.update(users).set({ verified: true }).where(eq(users.id, user.id));
  }

  // upsert tenant (link to user)
  let [tenant] = await db.select().from(tenants).where(eq(tenants.ownerUserId, user.id)).limit(1);
  if (!tenant){
    const id = crypto.randomUUID();
    await db.insert(tenants).values({ id, ownerUserId: user.id, name: 'Sweet Treats Bakery' });
    [tenant] = await db.select().from(tenants).where(eq(tenants.ownerUserId, user.id)).limit(1);
  }

  // upsert profile w/ slug + publish
  let [profile] = await db.select().from(tenantProfiles).where(eq(tenantProfiles.tenantId, tenant.id)).limit(1);
  const desired = { slug, isPublished: true, city: 'Bullard, TX', about: 'Custom cakes & desserts made fresh.', logoUrl: '/static/demo-logo.png' };
  if (!profile){
    await db.insert(tenantProfiles).values({ tenantId: tenant.id, ...desired });
  } else {
    await db.update(tenantProfiles).set({ ...desired }).where(eq(tenantProfiles.tenantId, tenant.id));
  }

  console.log('DEMO READY:', { email, password, slug });
}
run().catch(e=>{ console.error(e); process.exit(1); });
