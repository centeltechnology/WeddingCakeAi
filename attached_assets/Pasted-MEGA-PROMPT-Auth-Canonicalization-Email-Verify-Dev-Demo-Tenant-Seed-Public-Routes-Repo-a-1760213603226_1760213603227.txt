app.post('/api/auth/dev-verify', async (req,res)=>{
  if (process.env.DEMO_MODE !== 'true') return res.status(403).json({error:'forbidden'});
  const { email } = req.body ?? {};
  if (!email) return res.status(400).json({error:'email_required'});

  await usersRepo.markVerifiedByEmail(email);
  return res.json({ ok:true, verified:true });

E) Smoke Tests (fast feedback)

Create apps/app/scripts/smoke-auth-public.ts:

import fetch from 'node-fetch';
const base = 'http://localhost:5000';

async function check(p, m='GET'){ const r = await fetch(base+p, { method:m }); return {p, ok:r.ok, s:r.status}; }
(async()=>{
  const tests = [
    await check('/login'),                     // SPA served
    await check('/baker-login'),               // 301 -> /login
    await check('/signup'),                    // 301 -> /login?mode=signup
    await check('/p/sweet-treats-bakery'),     // 200 published listing
    await check('/b/sweet-treats-bakery/book'),// 200 booking page
    await check('/calculator?tenant=sweet-treats-bakery'), // 200 public calculator
  ];
  console.table(tests);
  const fail = tests.find(t=>!t.ok && t.s!==301);
  if (fail) process.exit(1);
  console.log('AUTH+PUBLIC ROUTES PASS ✅');
})();
