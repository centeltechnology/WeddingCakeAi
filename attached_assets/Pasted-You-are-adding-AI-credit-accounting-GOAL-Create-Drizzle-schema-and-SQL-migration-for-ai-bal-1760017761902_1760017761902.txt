CREATE TABLE IF NOT EXISTS ai_balances (
  id varchar PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id varchar NOT NULL UNIQUE,
  credits_balance integer NOT NULL DEFAULT 0,
  rollover_cap integer NOT NULL DEFAULT 0,
  cycle_start timestamp NOT NULL DEFAULT now(),
  cycle_end timestamp NOT NULL
);

CREATE TABLE IF NOT EXISTS ai_usage (
  id varchar PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id varchar NOT NULL,
  feature text NOT NULL,
  credits integer NOT NULL,
  tokens_input integer,
  tokens_output integer,
  unit_cost_cents integer DEFAULT 0,
  created_at timestamp NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS ai_usage_tenant_created_idx ON ai_usage (tenant_id, created_at);
