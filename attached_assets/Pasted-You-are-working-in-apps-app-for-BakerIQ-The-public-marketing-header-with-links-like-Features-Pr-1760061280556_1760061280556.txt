6) Static header strings scan (guard)

Add a tiny script apps/app/scripts/noMarketingHeaderInAdmin.ts that scans client/src/pages and fails if it finds the marketing labels in any file that also imports AppLayout.

// scripts/noMarketingHeaderInAdmin.ts
import fg from 'fast-glob';
import { readFileSync } from 'fs';
import { resolve, relative } from 'path';

const ROOT = resolve(process.cwd(), 'apps/app');
const pages = fg.sync('client/src/pages/**/*.tsx', { cwd: ROOT, absolute: true });

const BAD_STRINGS = ['Features','Pricing','Find Bakers','Support','Baker Login','Get Started','Live Demo'];
let bad: string[] = [];

for (const file of pages) {
  const src = readFileSync(file, 'utf8');
  const usesAppLayout = src.includes("from '@/components/AppLayout'") || src.includes('<AppLayout');
  if (!usesAppLayout) continue;
  if (BAD_STRINGS.some(s => src.includes(s))) {
    bad.push(relative(ROOT, file));
  }
}

if (bad.length) {
  console.error('Found marketing header strings inside admin pages:\n' + bad.map(f=>' - '+f).join('\n'));
  process.exit(1);
} else {
  console.log('PASS: no marketing header strings inside admin pages.');
}
