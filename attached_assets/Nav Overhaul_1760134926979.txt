D) Fix Logout (works for cookie or token)

Client util client/src/lib/auth.ts:

export const LOGOUT_PATH = '/logout'; // adjust here if needed
export async function logout(){
  // try server logout (for cookie sessions)
  try { await fetch(LOGOUT_PATH, { method:'POST', credentials:'include' }); } catch {}
  // client-side cleanup (for JWT/localStorage)
  try { localStorage.removeItem('token'); sessionStorage.removeItem('token'); } catch {}
  window.location.href = '/login';
}
