0) Server: derive tenant/baker in one place

Create apps/app/server/lib/context.ts:

export function getCtx(req:any){
  const tenantId = req?.tenant?.id;
  const userId = req?.user?.id ?? null;
  const bakerId = req?.user?.bakerId ?? null;
  if (!tenantId) throw new Error('No tenant in request');
  return { tenantId, userId, bakerId };
}

2) Server: Sample Quote seeder (only if no quotes)

Add to routes.ts:

app.post('/api/quotes/sample', ensureAuthUnified, async (req, res) => {
  const { tenantId } = getCtx(req);
  const hasQuote = await db.select({count: sql<number>`count(*)`.mapWith(Number)}).from(quotes)
    .where(eq(quotes.tenantId, tenantId));
  if ((hasQuote?.[0]?.count ?? 0) > 0) return res.json({ ok:true, created:false });

  // Ensure one customer exists
  let [cust] = await db.select().from(customers).where(eq(customers.tenantId, tenantId)).limit(1);
  if (!cust){
    const custId = crypto.randomUUID();
    await db.insert(customers).values({ id:custId, tenantId, name:'Sample Customer', email:'customer@example.com' });
    [cust] = await db.select().from(customers).where(eq(customers.id, custId));
  }

  const quoteId = crypto.randomUUID();
  await db.insert(quotes).values({
    id: quoteId, tenantId, customerId: cust.id, title: '8” Cake + 12 Cupcakes', status:'draft', total: 91,
    createdAt: new Date()
  });
  // Optional: items table if present
  try {
    await db.insert(quoteItems).values([
      { id: crypto.randomUUID(), tenantId, quoteId, name:'8” Two-Layer Cake', qty:1, price:55 },
      { id: crypto.randomUUID(), tenantId, quoteId, name:'Cupcakes (Dozen)', qty:1, price:36 },
    ]);
  } catch {}
  return res.json({ ok:true, created:true, quoteId });
});

import fetch from 'node-fetch';
const base = 'http://localhost:5000';
async function call(p:string, m='GET', b?:any){ const r = await fetch(base+p,{method:m, headers:{'Content-Type':'application/json'}, body:b?JSON.stringify(b):undefined}); return {p,ok:r.ok,status:r.status,txt:await r.text()} }
(async()=>{
  const out = [];
  out.push(await call('/api/leads/sample','POST'));
  out.push(await call('/api/quotes/sample','POST'));
  out.push(await call('/api/quotes'));
  // grab first quote id
  const q = JSON.parse((await call('/api/quotes')).txt)[0];
  if (q?.id){ out.push(await call(`/api/quotes/${q.id}/approve`,'POST')); }
  // list contracts
  out.push(await call('/api/contracts'));
  // sign first contract
  const c = JSON.parse((await call('/api/contracts')).txt)[0];
  if (c?.id){ out.push(await call(`/api/contracts/${c.id}/sign`,'POST')); }
  // mark first invoice paid
  const inv = JSON.parse((await call('/api/invoices')).txt)[0];
  if (inv?.id){ out.push(await call(`/api/invoices/${inv.id}/paid`,'POST')); }
  console.table(out.map(x=>({path:x.p,status:x.status,ok:x.ok})));
  const fails = out.filter(x=>!x.ok); if (fails.length){ console.error('FAIL', fails.map(f=>f.path)); process.exit(1); }
  console.log('QCI PASS ✅');
})();
