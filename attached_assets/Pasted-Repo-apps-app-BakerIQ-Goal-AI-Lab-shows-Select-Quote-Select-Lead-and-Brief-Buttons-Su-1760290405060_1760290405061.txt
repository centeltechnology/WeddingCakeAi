// Helper: ensure we have a quote for a given lead (create draft if needed)
async function ensureQuoteForLead(db: any, tenantId: string, leadId: string) {
  // 1) fetch lead
  const [lead] = await db.select().from(leads).where(and(eq(leads.id, leadId), eq(leads.tenantId, tenantId))).limit(1);
  if (!lead) throw new Error('lead_not_found');

  // 2) ensure customer exists
  let [customer] = await db.select().from(customers).where(and(eq(customers.tenantId, tenantId), eq(customers.id, lead.customerId))).limit(1);
  if (!customer) {
    // fallback upsert by email/phone if needed
    const cid = crypto.randomUUID();
    await db.insert(customers).values({
      id: cid, tenantId,
      name: lead.name ?? lead.email ?? lead.phone ?? 'Lead',
      email: lead.email ?? null,
      phone: lead.phone ?? null
    });
    [customer] = await db.select().from(customers).where(eq(customers.id, cid)).limit(1);
  }

  // 3) find existing draft quote for this customer, else create
  const [existing] = await db.select().from(quotes).where(
    and(eq(quotes.tenantId, tenantId), eq(quotes.customerId, customer.id), eq(quotes.status, 'draft'))
  ).limit(1);

  if (existing) return existing.id;

  const qid = crypto.randomUUID();
  await db.insert(quotes).values({
    id: qid, tenantId, customerId: customer.id,
    title: `Draft for ${customer.name ?? 'Lead'}`, status: 'draft',
    subtotal: 0, discount: 0, tax: 0, total: 0, depositPct: 0, createdAt: new Date()
  });

  try {
    await db.insert(quoteEvents).values({
      id: crypto.randomUUID(), tenantId, quoteId: qid, type: 'created',
      meta: { source: 'lead_bridge', leadId }, createdAt: new Date()
    });
  } catch {}

  return qid;
}

async function run(path: string, body: any){
  const r = await fetch(path, { method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify(body) });
  const j = await r.json().catch(()=> ({}));
  if (!r.ok) throw new Error(j?.error || `request_failed_${r.status}`);
  return j;
}

  const { items } = req.body ?? {};

  // verify quote belongs to tenant
  const [q] = await db.select().from(quotes).where(and(eq(quotes.id, quoteId), eq(quotes.tenantId, tenantId))).limit(1);
  if (!q) return res.status(404).json({ error: 'quote_not_found' });

  for (const it of items) {
    await db.insert(quoteItems).values({
      id: crypto.randomUUID(),
