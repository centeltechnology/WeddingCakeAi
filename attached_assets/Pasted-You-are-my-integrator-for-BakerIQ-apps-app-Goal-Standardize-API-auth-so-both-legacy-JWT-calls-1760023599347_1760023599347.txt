# 1) Create unified middleware

import type { Request, Response, NextFunction } from "express";
import jwt from "jsonwebtoken";

export function ensureAuthUnified(req: any, res: Response, next: NextFunction) {
  // Session first
  if (req.session?.userId) {
    req.user = {
      id: req.session.userId,
      role: req.session.role || "baker",
      tenantId: req.session.tenantId || req.session?.tenant_id || null,
    };
    return next();
  }
  // Legacy JWT (compat)
  const auth = req.headers?.authorization || "";
  if (auth.startsWith("Bearer ")) {
    try {
      const token = auth.slice(7);
      const payload: any = jwt.verify(token, process.env.JWT_SECRET || "dev");
      req.user = {
        id: payload.sub || payload.id,
        role: payload.role || "baker",
        tenantId: payload.tenantId || payload.tenant_id || null,
      };
      return next();
    } catch (e) {
      // fall through to 401
    }
  }
  return res.status(401).json({ ok: false, error: "Unauthenticated" });
}

export function requireTenant(req: any, res: Response, next: NextFunction) {
  if (!req.user?.tenantId) return res.status(400).json({ ok:false, error:"Missing tenant context" });
  return next();
}
