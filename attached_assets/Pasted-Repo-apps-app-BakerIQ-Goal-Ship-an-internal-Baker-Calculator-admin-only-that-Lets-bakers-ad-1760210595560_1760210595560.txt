2) Server endpoints (admin/internal)

Add to server/routes.ts (auth: ensureAuthUnified; context via req.tenant.id):

// List customers (for picker)
app.get('/api/customers', ensureAuthUnified, async (req,res)=>{
  const tenantId = req.tenant.id;
  const rows = await db.select().from(customers).where(eq(customers.tenantId, tenantId)).limit(100);
  res.json(rows);
});

// Create or upsert customer (by email or phone)
app.post('/api/customers/upsert', ensureAuthUnified, async (req,res)=>{
  const tenantId = req.tenant.id;
  const { name, email, phone } = req.body ?? {};
  if (!name && !email && !phone) return res.status(400).json({error:'invalid_customer'});
  // prefer email match, else phone
  let row = null;
  if (email) [row] = await db.select().from(customers).where(and(eq(customers.tenantId, tenantId), eq(customers.email, email))).limit(1);
  if (!row && phone) [row] = await db.select().from(customers).where(and(eq(customers.tenantId, tenantId), eq(customers.phone, phone))).limit(1);
  if (row) {
    await db.update(customers).set({ name: name ?? row.name, phone: phone ?? row.phone }).where(eq(customers.id, row.id));
  } else {
    const id = crypto.randomUUID();
    await db.insert(customers).values({ id, tenantId, name: name ?? email ?? phone, email: email ?? null, phone: phone ?? null });
    [row] = await db.select().from(customers).where(eq(customers.id, id));
  }
  res.json({ ok:true, customer: row });
});

// Create quote w/ items from an "estimate"
app.post('/api/estimates/save-as-quote', ensureAuthUnified, async (req,res)=>{
  const tenantId = req.tenant.id;
  const { customer, title, items, taxRate, discount, depositPct, notes } = req.body ?? {};
  if (!Array.isArray(items) || items.length===0) return res.status(400).json({error:'no_items'});
  // Upsert or use customerId
  let customerId = customer?.id ?? null;
  if (!customerId) {
    const up = await fetchLike('/api/customers/upsert', { method:'POST', body: { name: customer?.name, email: customer?.email, phone: customer?.phone } });
    customerId = up.customer.id;
  }
  // compute totals
  const subtotal = items.reduce((s,it)=> s + (Number(it.qty)||0) * (Number(it.price)||0), 0);
  const discountAmt = Math.max(0, Number(discount)||0);
  const taxed = Math.max(0, (Number(taxRate)||0)) * Math.max(0, subtotal - discountAmt);
  const total = Math.max(0, subtotal - discountAmt + taxed);
  const depositPctN = Math.max(0, Math.min(1, Number(depositPct) || 0));

  const quoteId = crypto.randomUUID();
  await db.insert(quotes).values({
    id: quoteId, tenantId, customerId, title: title || 'Estimate', status:'draft',
    subtotal, discount: discountAmt, tax: taxed, total, depositPct: depositPctN, notes: notes ?? null, createdAt: new Date()
  });

  // items
  for (const it of items){
    await db.insert(quoteItems).values({
      id: crypto.randomUUID(),
      tenantId, quoteId,
      name: String(it.name||'Item'),
      qty: Number(it.qty)||1,
      unit: it.unit ?? 'ea',
      price: Number(it.price)||0,
      notes: it.notes ?? null,
    });
  }
  // event
  try { await db.insert(quoteEvents).values({ id: crypto.randomUUID(), tenantId, quoteId, type:'created', meta: { source:'baker_calculator' }, createdAt: new Date() }); } catch {}
  res.status(201).json({ ok:true, quoteId, totals: { subtotal, discount: discountAmt, tax: taxed, total, depositPct: depositPctN } });
});


fetchLike above is pseudo; in your file, call the upsert function directly or extract a service util—goal is atomic creation.
