# =========================
# 1) Server: cookie strategy + /api/session no-store
# =========================
Open apps/app/server/index.ts and update the session config:

- Compute crossSite:
  const crossSite = process.env.CROSS_SITE_COOKIES === "true";
- Use:
  app.use(session({
    name: "sid",
    store: new PgSession({ ...existing... }),
    secret: process.env.SESSION_SECRET || "fallback_session_secret_for_development_only",
    resave: false,
    saveUninitialized: false,
    cookie: {
      httpOnly: true,
      maxAge: 7 * 24 * 60 * 60 * 1000,
      sameSite: crossSite ? "none" : "lax",
      secure: crossSite || process.env.NODE_ENV === "production"
    }
  }));

- Add no-store headers on the session probe endpoint:
  app.get("/api/session", (req, res) => {
    res.set("Cache-Control", "no-store");
    const authed = Boolean((req.session as any)?.userId);
    res.json({
      authenticated: authed,
      userId: authed ? (req.session as any).userId : null,
      role: (req.session as any)?.role || null,
      isImpersonating: Boolean((req.session as any)?.isImpersonating) || false
    });
  });

# =========================
# 2) Client: smart root redirect + always include credentials + no-store
# =========================
Edit apps/app/client/src/App.tsx:
- Replace the "/" route logic. Instead of hard Redirect to "/login", create a small component that checks /api/session and redirects accordingly.

Create (or inline) a RootGate component:

-----------------------------------------
import { useEffect } from "react";
import { useLocation } from "wouter";

function RootGate() {
  const [, navigate] = useLocation();
  useEffect(() => {
    (async () => {
      try {
        const res = await fetch("/api/session", { credentials: "include", cache: "no-store" });
        const data = await res.json();
        const target = data?.authenticated ? "/baker/dashboard" : "/login";
        navigate(target, { replace: true });
      } catch {
        navigate("/login", { replace: true });
      }
    })();
  }, [navigate]);
  return null;
}
-----------------------------------------

In the router:
- Change the root route to: <Route path="/"><RootGate /></Route>
- Keep all other routes the same; ensure protected routes still use <AuthGuard>.

Ensure every client fetch that checks auth/session uses:
{ credentials: "include", cache: "no-store" }

# =========================
# 3) (Optional but recommended) Vite dev proxy for same-origin API
# =========================
If not already present, add a dev proxy so the browser sees /api as same-origin, which avoids cross-site cookie rules entirely.

Edit/create apps/app/client/vite.config.ts:

-----------------------------------------
import { defineConfig } from "vite";
import react from "@vitejs/plugin-react";

export default defineConfig({
  plugins: [react()],
  server: {
    proxy: {
      "/api": {
        target: "http://localhost:3000",    // adjust if your Express runs elsewhere
        changeOrigin: true,
        secure: false
      }
    }
  }
});
-----------------------------------------

If using a different port for the API, update target accordingly.
After adding a proxy, you may set CROSS_SITE_COOKIES=false to use SameSite=lax and secure=false in dev.
