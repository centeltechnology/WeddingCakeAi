## POSTGRES BLOCK (app)
BEGIN;

-- 1) Tenancy integrity: no NULL tenant_id in key tables
WITH checks AS (
  SELECT 'bookings'::text AS t, COUNT(*) c FROM bookings WHERE tenant_id IS NULL
  UNION ALL SELECT 'consultations', COUNT(*) FROM consultations WHERE tenant_id IS NULL
  UNION ALL SELECT 'messages', COUNT(*) FROM messages WHERE tenant_id IS NULL
  UNION ALL SELECT 'analytics', COUNT(*) FROM analytics WHERE tenant_id IS NULL
  UNION ALL SELECT 'reviews', COUNT(*) FROM reviews WHERE tenant_id IS NULL
)
SELECT * FROM checks;

-- 2) Quote → Lead integrity: forbid NULL lead_id
SELECT COUNT(*) AS quotes_without_lead FROM quotes WHERE lead_id IS NULL;

-- 3) Email uniqueness (app-side): dup emails in bakers/customers
SELECT email, COUNT(*) c FROM bakers GROUP BY email HAVING COUNT(*) > 1;
SELECT email, COUNT(*) c FROM customers GROUP BY email HAVING COUNT(*) > 1;

-- 4) Payments surface check
SELECT COUNT(*) AS invoices_without_valid_contract
FROM invoices i LEFT JOIN contracts c ON c.id = i.contract_id
WHERE i.contract_id IS NOT NULL AND c.id IS NULL;

SELECT COUNT(*) AS transactions_without_intent
FROM transactions
WHERE stripe_payment_intent_id IS NULL;

-- 5) Activity logs FK sanity
SELECT COUNT(*) AS activity_orphans
FROM activity_logs al LEFT JOIN tenants t ON t.id = al.tenant_id
WHERE al.tenant_id IS NOT NULL AND t.id IS NULL;

-- 6) Phase 3 spot checks (from earlier)
SELECT column_name, data_type
FROM information_schema.columns
WHERE table_name='customers' AND column_name='address_json';

SELECT column_name, data_type
FROM information_schema.columns
WHERE table_name='reviews' AND column_name='contract_id';

-- 7) Optional FK check (reviews → contracts)
SELECT conname
FROM pg_constraint
WHERE conrelid = 'reviews'::regclass AND conname='fk_reviews_contract';

COMMIT;
